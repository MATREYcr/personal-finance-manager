import React from 'react';

/**
 * Badge — small status/category pill with an optional leading status dot,
 * matching the reference's Active/Paused row-status treatment.
 */
export function Badge({ variant = 'default', dot = false, children, style }) {
  const variants = {
    default: { background: 'var(--accent-subtle)', color: 'var(--accent)', border: 'none' },
    positive: { background: 'var(--positive-bg)', color: 'var(--positive)', border: '1px solid var(--positive-border)' },
    negative: { background: 'var(--negative-bg)', color: 'var(--negative)', border: '1px solid var(--negative-border)' },
    warning: { background: 'var(--warning-bg)', color: 'var(--warning)', border: '1px solid var(--warning-border)' },
    outline: { background: 'transparent', color: 'var(--text-secondary)', border: '1px solid var(--border-default)' },
  };
  const dotColors = {
    default: 'var(--accent)',
    positive: 'var(--positive)',
    negative: 'var(--negative)',
    warning: 'var(--warning)',
    outline: 'var(--text-muted)',
  };
  return (
    <span
      style={{
        display: 'inline-flex',
        alignItems: 'center',
        gap: 6,
        height: 22,
        padding: dot ? '0 10px 0 8px' : '0 10px',
        borderRadius: 'var(--radius-full)',
        fontSize: 'var(--text-sm)',
        fontWeight: 600,
        fontFamily: 'var(--font-heading)',
        ...variants[variant],
        ...style,
      }}
    >
      {dot && <span style={{ width: 6, height: 6, borderRadius: 'var(--radius-full)', background: dotColors[variant], flexShrink: 0 }} />}
      {children}
    </span>
  );
}
