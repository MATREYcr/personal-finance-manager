const { useState } = React;

const initialCategories = [
  { id: 1, name: 'Market', type: 'Expense' },
  { id: 2, name: 'Transport', type: 'Expense' },
  { id: 3, name: 'Rent', type: 'Expense' },
  { id: 4, name: 'Utilities', type: 'Expense' },
  { id: 5, name: 'Entertainment', type: 'Expense' },
  { id: 6, name: 'Health', type: 'Expense' },
  { id: 7, name: 'Other', type: 'Expense' },
  { id: 8, name: 'Salary', type: 'Income' },
  { id: 9, name: 'Other Income', type: 'Income' },
];

function Categories({ Button, Input, Label, Select, Badge, Dialog, Table, TableHead, TableHeadRow, TableHeaderCell, TableBody, TableRow, TableCell, errorDefaultOpen }) {
  const [dialogOpen, setDialogOpen] = useState(false);
  const [formType, setFormType] = useState('expense');
  const [deleteErrorId, setDeleteErrorId] = useState(errorDefaultOpen ? 1 : null);

  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 20, maxWidth: 'var(--content-max-width)' }}>
      <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', flexWrap: 'wrap', gap: 12 }}>
        <h1 style={{ fontSize: 'var(--text-2xl)', fontWeight: 700 }}>Categories</h1>
        <Button onClick={() => setDialogOpen(true)}>
          <i data-lucide="plus" style={{ width: 16, height: 16 }}></i>
          New category
        </Button>
      </div>

      <Table>
        <TableHead><TableHeadRow>
          <TableHeaderCell>Name</TableHeaderCell>
          <TableHeaderCell>Type</TableHeaderCell>
          <TableHeaderCell align="right">Actions</TableHeaderCell>
        </TableHeadRow></TableHead>
        <TableBody>
          {initialCategories.map((c) => (
            <React.Fragment key={c.id}>
              <TableRow style={deleteErrorId === c.id ? { borderBottom: 'none' } : {}}>
                <TableCell>{c.name}</TableCell>
                <TableCell><Badge variant={c.type === 'Income' ? 'positive' : 'outline'} dot={c.type === 'Income'}>{c.type}</Badge></TableCell>
                <TableCell align="right">
                  <div style={{ display: 'flex', gap: 4, justifyContent: 'flex-end' }}>
                    <Button variant="default" size="icon" aria-label="Edit" style={{ background: 'var(--accent)', borderColor: 'var(--accent)', color: '#fff' }}><i data-lucide="pencil" style={{ width: 15, height: 15 }}></i></Button>
                    <Button variant="default" size="icon" aria-label="Delete" style={{ background: 'var(--negative)', borderColor: 'var(--negative)', color: '#fff' }} onClick={() => setDeleteErrorId(deleteErrorId === c.id ? null : c.id)}>
                      <i data-lucide="trash-2" style={{ width: 15, height: 15 }}></i>
                    </Button>
                  </div>
                </TableCell>
              </TableRow>
              {deleteErrorId === c.id && (
                <tr>
                  <td colSpan={3} style={{ padding: '0 16px 14px' }}>
                    <div style={{ display: 'flex', gap: 10, alignItems: 'flex-start', background: 'var(--negative-bg)', border: '1px solid var(--negative-300)', borderRadius: 'var(--radius-md)', padding: '10px 14px' }}>
                      <i data-lucide="alert-circle" style={{ width: 16, height: 16, color: 'var(--negative-strong)', flexShrink: 0, marginTop: 2 }}></i>
                      <span style={{ fontSize: 'var(--text-sm)', color: 'var(--negative-strong)' }}>
                        Cannot delete a category that still has transactions. Reassign or delete those transactions first.
                      </span>
                    </div>
                  </td>
                </tr>
              )}
            </React.Fragment>
          ))}
        </TableBody>
      </Table>

      <Dialog
        open={dialogOpen}
        onClose={() => setDialogOpen(false)}
        title="New category"
        footer={<><Button variant="outline" onClick={() => setDialogOpen(false)}>Cancel</Button><Button onClick={() => setDialogOpen(false)}>Save</Button></>}
      >
        <div style={{ display: 'flex', flexDirection: 'column', gap: 14 }}>
          <div>
            <Label>Name</Label>
            <Input placeholder="e.g. Groceries" />
          </div>
          <div>
            <Label>Type</Label>
            <Select value={formType} onChange={setFormType} options={[{ value: 'expense', label: 'Expense' }, { value: 'income', label: 'Income' }]} />
          </div>
        </div>
      </Dialog>
    </div>
  );
}
window.Categories = Categories;
